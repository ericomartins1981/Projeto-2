import numpy as np
import matplotlib.pyplot as plt

N = 300_000                  
snr_media_db = 10                    
snr_media_lin = 10**(snr_media_db / 10)  
L_valores = [1, 2, 3, 4]             

gamma_th_db = np.linspace(-10, 25, 100)
gamma_th_lin = 10**(gamma_th_db / 10)

plt.figure(figsize=(9, 6))
cores = ['#1f77b4', '#ff7f0e', '#2ca02c', '#d62728']
marcadores = ['+', '+', '+', '+']

rng = np.random.default_rng(seed=42)

for idx, L in enumerate(L_valores):
    potencia_canal = rng.exponential(scale=1.0, size=(N, L))
    gamma_l = snr_media_lin * potencia_canal
    
    gamma_s_egc = (1.0 / L) * (np.sum(np.sqrt(gamma_l), axis=1))**2
    
    op = np.mean(gamma_s_egc[:, np.newaxis] < gamma_th_lin[np.newaxis, :], axis=0)
    
    plt.semilogy(
        gamma_th_db, 
        op, 
        label=f'L = {L}', 
        color=cores[idx], 
        marker=marcadores[idx], 
        markevery=5, 
        linewidth=0, 
        markersize=10
    )

plt.title(f'Probabilidade de Interrupção (OP) - Combinador EGC ($\\bar{{\\gamma}} = {snr_media_db}$ dB, $N = {N}$)', fontsize=12)
plt.xlabel('Limiar de SNR $\\gamma_{\\text{th}}$ (dB)', fontsize=11)
plt.ylabel('Probabilidade de Interrupção ($OP$)', fontsize=11)
plt.ylim([1e-3, 1])
plt.xlim([gamma_th_db[0], gamma_th_db[-1]])
plt.grid(True, which='both', linestyle='--', alpha=0.6)
plt.legend(title='Ordem de Diversidade', fontsize=10)
plt.tight_layout()
plt.show()
