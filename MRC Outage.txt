import numpy as np
import matplotlib.pyplot as plt
from scipy.special import factorial

def mrc_outage_probability(gamma_th, gamma_bar, L):
    gamma_th = np.maximum(gamma_th, 1e-10)
    op = np.zeros_like(gamma_th, dtype=float)
    
    for idx, g_th in enumerate(gamma_th):
        sum_term = 0.0
        for k in range(L):
            sum_term += (1 / factorial(k)) * (g_th / gamma_bar) ** k
        op[idx] = 1 - np.exp(-g_th / gamma_bar) * sum_term
        
    return np.clip(op, 0, 1)

def monte_carlo_mrc_outage(gamma_th_list, gamma_bar, L, num_samples=300000):
    op_sim = []
    for g_th in gamma_th_list:
        gamma_branches = np.random.exponential(scale=gamma_bar, size=(num_samples, L))
        gamma_mrc = np.sum(gamma_branches, axis=1)
        outages = np.sum(gamma_mrc < g_th)
        op_sim.append(outages / num_samples)
    return np.array(op_sim)

gamma_bar_dB = 10.0
gamma_bar_linear = 10 ** (gamma_bar_dB / 10)

gamma_th_db = np.linspace(-5, 20, 150)
gamma_th_linear = 10 ** (gamma_th_db / 10)

gamma_th_sim_db = np.array([-5, 0, 5, 10, 15, 20])
gamma_th_sim_linear = 10 ** (gamma_th_sim_db / 10)

L_values = [1, 2, 3, 4]
colors = ['#1f77b4', '#ff7f0e', '#2ca02c', '#d62728']

plt.figure(figsize=(10, 6))

for i, L in enumerate(L_values):
    op_theory = mrc_outage_probability(gamma_th_linear, gamma_bar_linear, L)
    plt.semilogy(gamma_th_db, op_theory, color=colors[i], linestyle='-', linewidth=2, 
                 label=f'Teoria: L = {L}')
    
    op_sim = monte_carlo_mrc_outage(gamma_th_sim_linear, gamma_bar_linear, L, num_samples=300000)
    plt.semilogy(gamma_th_sim_db, op_sim, color='black', marker='x', linestyle='None', 
                 markersize=7, markeredgewidth=1.5, label=f'Simulação: L = {L}')

plt.xlabel('Limiar de SNR $\\gamma_{\\text{th}}$ [dB] ($\\bar{\\gamma}_s = 10$ dB)', fontsize=12)
plt.ylabel('Probabilidade de Interrupção ($P_{\\text{out}}$)', fontsize=12)
plt.title('Outage Probability (OP), Maximal Ratio Combining (MRC) sob Rayleigh', fontsize=13)
plt.grid(True, which="both", ls="--", alpha=0.5)
plt.legend(fontsize=10, ncol=2)
plt.ylim(1e-5, 1)
plt.xlim(-5, 25)

plt.tight_layout()
plt.show()
