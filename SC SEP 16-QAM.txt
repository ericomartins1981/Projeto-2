import numpy as np
import matplotlib.pyplot as plt
from scipy.special import comb, erfc

def sc_sep_from_image(gamma_bar, L, M=16):
    a_M = 4 * (np.sqrt(M) - 1) / np.sqrt(M)
    b_M = 3 / (M - 1)
    
    gamma_bar = np.maximum(gamma_bar, 1e-10)
    sep_total = np.zeros_like(gamma_bar, dtype=float)
    
    for l in range(L):
        binom_coeff = comb(L - 1, l)
        sign = (-1) ** l
        factor = L / (2 * (l + 1))
        
        numerator = (b_M * gamma_bar) / (2 * (l + 1))
        term_sqrt = np.sqrt(numerator / (1 + numerator))
        inner_expr = 1 - term_sqrt
        
        sep_total += binom_coeff * sign * factor * inner_expr
        
    return a_M * sep_total

def monte_carlo_sc_sep(gamma_bar_list, L, M=16, num_samples=200000):
    a_M = 4 * (np.sqrt(M) - 1) / np.sqrt(M)
    b_M = 3 / (M - 1)
    sep_sim = []
    
    for g_bar in gamma_bar_list:
        if g_bar <= 0:
            sep_sim.append(1.0)
            continue
            
        u = np.random.uniform(0, 1, num_samples)
        gamma_sc = -g_bar * np.log(1 - u ** (1 / L))
        
        arg = np.sqrt(b_M * gamma_sc)
        q_vals = 0.5 * erfc(arg / np.sqrt(2))
        pe_awgn = a_M * q_vals - ((np.sqrt(M) - 1) / np.sqrt(M)) ** 2 * (q_vals ** 2)
        
        sep_sim.append(np.mean(pe_awgn))
        
    return np.array(sep_sim)

M = 16
snr_db = np.array([0, 5, 10, 15, 20, 25, 30])
snr_db_theory = np.linspace(0, 30, 150)
snr_linear_theory = 10 ** (snr_db_theory / 10)
snr_linear_sim = 10 ** (snr_db / 10)

L_values = [1, 2, 3, 4]
colors = ['#1f77b4', '#ff7f0e', '#2ca02c', '#d62728']

plt.figure(figsize=(10, 6))

for i, L in enumerate(L_values):
    sep_theory = sc_sep_from_image(snr_linear_theory, L, M)
    plt.semilogy(snr_db_theory, sep_theory, color=colors[i], linestyle='-', linewidth=2, 
                 label=f'Teoria: L = {L}')
    
    sep_sim = monte_carlo_sc_sep(snr_linear_sim, L, M, num_samples=300000)
    plt.semilogy(snr_db, sep_sim, color='black', marker='x', linestyle='None', 
                 markersize=7, markeredgewidth=1.5, label=f'Simulação: L = {L}')

plt.xlabel('SNR Média por Ramo ($\\bar{\gamma}_s$) [dB]', fontsize=12)
plt.ylabel('Probabilidade Média de Erro de Símbolo ($\\bar{P}_s^{\\text{SC}}$)', fontsize=12)
plt.title('SEP, Selection Combining (SC) 16-QAM', fontsize=13)
plt.grid(True, which="both", ls="--", alpha=0.5)
plt.legend(fontsize=10, ncol=2)
plt.ylim(1e-5, 1)
plt.xlim(0, 30)

plt.tight_layout()
plt.show()
