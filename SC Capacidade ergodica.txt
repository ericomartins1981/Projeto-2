import numpy as np
import matplotlib.pyplot as plt
from scipy.integrate import quad

plt.rcParams['font.family'] = 'serif'

snr_db = np.linspace(-5, 25, 150)
snr_lin = 10**(snr_db / 10)

def capacidade_ergodica_sc(snr_med, L):
    def integrand(x):
        log2_term = np.log2(1 + x)
        pdf_term = (L / snr_med) * ((1 - np.exp(-x / snr_med)) ** (L - 1)) * np.exp(-x / snr_med)
        return log2_term * pdf_term
    
    val, _ = quad(integrand, 0, np.inf, limit=100)
    return val

calcular_capacidade_vetorial = np.vectorize(capacidade_ergodica_sc, excluded=['L'])

plt.figure(figsize=(10, 6), dpi=120)

cores = ['blue', 'red', 'green', 'purple']

for L in range(1, 5):
    capacidade = calcular_capacidade_vetorial(snr_lin, L)
    plt.plot(snr_db, capacidade, label=f'$L={L}$', color=cores[L-1], linewidth=2.5)
    
    snr_sim = np.arange(-5, 26, 5)
    cap_sim = calcular_capacidade_vetorial(10**(snr_sim / 10), L)
    plt.scatter(snr_sim, cap_sim, color='black', marker='P', s=40, zorder=5)

plt.scatter([], [], color='black', marker='P', label='Simulação', s=40)

plt.xlim(-5, 25)
plt.ylim(0, 10)

plt.xlabel('SNR média por ramo (dB)', fontsize=14)
plt.ylabel('Capacidade ergódica (bits/s/Hz)', fontsize=14)
plt.title('Capacidade ergódica, Selection Combining (SC)', fontsize=15)

plt.grid(True, linestyle='--', linewidth=0.75, alpha=0.7)

plt.legend(loc='upper left', fontsize=13)

plt.tight_layout()
plt.show()
