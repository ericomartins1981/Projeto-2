import numpy as np
import matplotlib.pyplot as plt

N_samples = 300000
L = 3
K_ric = 5.0
snr_db = np.linspace(0, 30, 31)
snr_lin = 10**(snr_db / 10.0)

gain_sc_ray = np.zeros_like(snr_db, dtype=float)
gain_mrc_ray = np.zeros_like(snr_db, dtype=float)
gain_egc_ray = np.zeros_like(snr_db, dtype=float)

gain_sc_ric = np.zeros_like(snr_db, dtype=float)
gain_mrc_ric = np.zeros_like(snr_db, dtype=float)
gain_egc_ric = np.zeros_like(snr_db, dtype=float)

for i, gamma_bar in enumerate(snr_lin):
    
    gamma_ray = np.random.exponential(scale=gamma_bar, size=(N_samples, L))
    
    C_1_ray = np.mean(np.log2(1 + gamma_ray[:, 0]))
    C_sc_ray = np.mean(np.log2(1 + np.max(gamma_ray, axis=1)))
    C_mrc_ray = np.mean(np.log2(1 + np.sum(gamma_ray, axis=1)))
    C_egc_ray = np.mean(np.log2(1 + (np.sum(np.sqrt(gamma_ray), axis=1)**2) / L))
    
    gain_sc_ray[i] = 10 * np.log10(C_sc_ray / C_1_ray)
    gain_mrc_ray[i] = 10 * np.log10(C_mrc_ray / C_1_ray)
    gain_egc_ray[i] = 10 * np.log10(C_egc_ray / C_1_ray)
    
    g_real = np.random.normal(0, 1/np.sqrt(2), size=(N_samples, L))
    g_imag = np.random.normal(0, 1/np.sqrt(2), size=(N_samples, L))
    
    h_real = np.sqrt(K_ric / (K_ric + 1)) + np.sqrt(1 / (K_ric + 1)) * g_real
    h_imag = np.sqrt(1 / (K_ric + 1)) * g_imag
    
    h_sq = h_real**2 + h_imag**2
    gamma_ric = h_sq * gamma_bar
    
    C_1_ric = np.mean(np.log2(1 + gamma_ric[:, 0]))
    C_sc_ric = np.mean(np.log2(1 + np.max(gamma_ric, axis=1)))
    C_mrc_ric = np.mean(np.log2(1 + np.sum(gamma_ric, axis=1)))
    C_egc_ric = np.mean(np.log2(1 + (np.sum(np.sqrt(gamma_ric), axis=1)**2) / L))
    
    gain_sc_ric[i] = 10 * np.log10(C_sc_ric / C_1_ric)
    gain_mrc_ric[i] = 10 * np.log10(C_mrc_ric / C_1_ric)
    gain_egc_ric[i] = 10 * np.log10(C_egc_ric / C_1_ric)

plt.figure(figsize=(10, 7))

plt.plot(snr_db, gain_mrc_ray, 'r-', linewidth=2, label='MRC - Rayleigh')
plt.plot(snr_db, gain_egc_ray, 'g-', linewidth=2, label='EGC - Rayleigh')
plt.plot(snr_db, gain_sc_ray, 'b-', linewidth=2, label='SC - Rayleigh')

plt.plot(snr_db, gain_mrc_ric, 'r--', linewidth=2, label='MRC - Rice ($K=5$)')
plt.plot(snr_db, gain_egc_ric, 'g--', linewidth=2, label='EGC - Rice ($K=5$)')
plt.plot(snr_db, gain_sc_ric, 'b--', linewidth=2, label='SC - Rice ($K=5$)')

plt.title(f'Ganho de Diversidade ($L = {L}$) - Rayleigh e Rice ($K=5$)', fontsize=14)
plt.xlabel('SNR Média por Ramo $\\bar{\\gamma}$ (dB)', fontsize=12)
plt.ylabel('Ganho $G_l = 10 \\log_{10}(C_{\\text{erg}}(L) / C_{\\text{erg}}(1))$ (dB)', fontsize=12)
plt.grid(True, which='both', linestyle=':', alpha=0.7)
plt.legend(fontsize=10, loc='best')
plt.tight_layout()

plt.show()
