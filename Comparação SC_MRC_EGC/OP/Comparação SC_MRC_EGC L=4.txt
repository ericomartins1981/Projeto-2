import numpy as np
import matplotlib.pyplot as plt

N = 300_000
snr_media_db = 10
snr_media_lin = 10**(snr_media_db / 10)
L = 4

gamma_th_db = np.linspace(-10, 25, 100)
gamma_th_lin = 10**(gamma_th_db / 10)

plt.figure(figsize=(9, 6))
rng = np.random.default_rng(seed=42)

potencia_canal = rng.exponential(scale=1.0, size=(N, L))
gamma_l = snr_media_lin * potencia_canal

gamma_sc = np.max(gamma_l, axis=1)
gamma_mrc = np.sum(gamma_l, axis=1)
gamma_egc = (1.0 / L) * (np.sum(np.sqrt(gamma_l), axis=1))**2

tecnicas = {
    'SC': gamma_sc,
    'EGC': gamma_egc,
    'MRC': gamma_mrc
}

cores = {'SC': '#1f77b4', 'EGC': '#ff7f0e', 'MRC': '#2ca02c'}
marcadores = {'SC': '+', 'EGC': '+', 'MRC': '+'}

for nome, gamma_s in tecnicas.items():
    op = np.mean(gamma_s[:, np.newaxis] < gamma_th_lin[np.newaxis, :], axis=0)
    
    plt.semilogy(
        gamma_th_db,
        op,
        label=f'{nome} (L = {L})',
        color=cores[nome],
        marker=marcadores[nome],
        markevery=10,
        markersize=6,
        linewidth=0 if nome == 'EGC' else 1.5
    )

plt.title(f'Comparação OP: SC, EGC e MRC ($\\bar{{\\gamma}} = {snr_media_db}$ dB, $L = {L}$, $N = {N}$)', fontsize=12)
plt.xlabel('Limiar de SNR $\\gamma_{\\text{th}}$ (dB)', fontsize=11)
plt.ylabel('Probabilidade de Outage ($OP$)', fontsize=11)
plt.ylim([1e-3, 1])
plt.xlim([gamma_th_db[0], gamma_th_db[-1]])
plt.grid(True, which='both', linestyle='--', alpha=0.6)
plt.legend(fontsize=10)
plt.tight_layout()
plt.show()
