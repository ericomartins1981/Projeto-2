import numpy as np
import matplotlib.pyplot as plt
from scipy.special import comb, erfc

def mrc_sep_from_image(gamma_bar, L, M=16):
    a_M = 4 * (np.sqrt(M) - 1) / np.sqrt(M)
    b_M = 3 / (M - 1)
    
    gamma_bar = np.maximum(gamma_bar, 1e-10)
    
    numerator = (b_M * gamma_bar) / 2
    mu = np.sqrt(numerator / (1 + numerator))
    
    sep_total = np.zeros_like(gamma_bar, dtype=float)
    
    for k in range(L):
        binom_coeff = comb(L - 1 + k, k)
        term1 = ((1 - mu) / 2) ** L
        term2 = ((1 + mu) / 2) ** k
        sep_total += binom_coeff * term1 * term2
        
    return a_M * sep_total

def monte_carlo_mrc_sep(gamma_bar_list, L, M=16, num_samples=200000):
    a_M = 4 * (np.sqrt(M) - 1) / np.sqrt(M)
    b_M = 3 / (M - 1)
    sep_sim = []
    
    for g_bar in gamma_bar_list:
        if g_bar <= 0:
            sep_sim.append(1.0)
            continue
            
        gamma_branches = np.random.exponential(scale=g_bar, size=(num_samples, L))
        gamma_mrc = np.sum(gamma_branches, axis=1)
        
        arg = np.sqrt(b_M * gamma_mrc)
        q_vals = 0.5 * erfc(arg / np.sqrt(2))
        pe_awgn = a_M * q_vals - ((np.sqrt(M) - 1) / np.sqrt(M)) ** 2 * (q_vals ** 2)
        
        sep_sim.append(np.mean(pe_awgn))
        
    return np.array(sep_sim)

M = 16
snr_db = np.array([0, 5, 10, 15, 20, 25, 30])
snr_db_theory = np.linspace(0, 30, 150)
snr_linear_theory = 10 ** (snr_db_theory / 10)
snr_linear_sim = 10 ** (snr_db / 10)

L_values = [1, 2, 3, 4]
colors = ['#1f77b4', '#ff7f0e', '#2ca02c', '#d62728']

plt.figure(figsize=(10, 6))

for i, L in enumerate(L_values):
    sep_theory = mrc_sep_from_image(snr_linear_theory, L, M)
    plt.semilogy(snr_db_theory, sep_theory, color=colors[i], linestyle='-', linewidth=2, 
                 label=f'Teoria: L = {L}')
    
    sep_sim = monte_carlo_mrc_sep(snr_linear_sim, L, M, num_samples=300000)
    plt.semilogy(snr_db, sep_sim, color='black', marker='x', linestyle='None', 
                 markersize=7, markeredgewidth=1.5, label=f'Simulação: L = {L}')

plt.xlabel('SNR Média por Ramo ($\\bar{\\gamma}_s$) [dB]', fontsize=12)
plt.ylabel('Probabilidade Média de Erro de Símbolo ($\\bar{P}_s^{\\text{MRC}}$)', fontsize=12)
plt.title('SEP, Maximal Ratio Combining (MRC) 16-QAM sob Rayleigh', fontsize=13)
plt.grid(True, which="both", ls="--", alpha=0.5)
plt.legend(fontsize=10, ncol=2)
plt.ylim(1e-5, 1)
plt.xlim(0, 30)

plt.tight_layout()
plt.show()
