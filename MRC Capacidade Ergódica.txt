import numpy as np
import matplotlib.pyplot as plt
from scipy.special import expn

def mrc_ergodic_capacity(gamma_bar, L, Bw=1.0):
    gamma_bar = np.maximum(gamma_bar, 1e-10)
    capacity = np.zeros_like(gamma_bar, dtype=float)
    
    for idx, g_bar in enumerate(gamma_bar):
        sum_term = 0.0
        for k in range(1, L + 1):
            order = L - k + 1
            sum_term += expn(order, 1.0 / g_bar)
            
        capacity[idx] = (Bw / np.log(2)) * np.exp(1.0 / g_bar) * sum_term
        
    return capacity

def monte_carlo_mrc_capacity(gamma_bar_list, L, Bw=1.0, num_samples=300000):
    cap_sim = []
    for g_bar in gamma_bar_list:
        if g_bar <= 0:
            cap_sim.append(0.0)
            continue
            
        gamma_branches = np.random.exponential(scale=g_bar, size=(num_samples, L))
        gamma_mrc = np.sum(gamma_branches, axis=1)
        
        instantaneous_cap = (Bw / np.log(2)) * np.log2(1 + gamma_mrc)
        cap_sim.append(np.mean(instantaneous_cap))
        
    return np.array(cap_sim)

Bw = 1.0
snr_db_theory = np.linspace(-5, 20, 150)
snr_linear_theory = 10 ** (snr_db_theory / 10)

snr_db = np.array([-5, 0, 5, 10, 15, 20])
snr_linear_sim = 10 ** (snr_db / 10)

L_values = [1, 2, 3, 4]
colors = ['#1f77b4', '#ff7f0e', '#2ca02c', '#d62728']

plt.figure(figsize=(10, 6))

for i, L in enumerate(L_values):
    cap_theory = mrc_ergodic_capacity(snr_linear_theory, L, Bw)
    plt.plot(snr_db_theory, cap_theory, color=colors[i], linestyle='-', linewidth=2, 
             label=f'Teoria: L = {L}')
    
    cap_sim = monte_carlo_mrc_capacity(snr_linear_sim, L, Bw, num_samples=300000)
    plt.plot(snr_db, cap_sim, color='black', marker='x', linestyle='None', 
             markersize=7, markeredgewidth=1.5, label=f'Simulação: L = {L}')

plt.xlabel('SNR Média por Ramo ($\\bar{\\gamma}$) [dB]', fontsize=12)
plt.ylabel('Capacidade Ergódica ($C_{\\text{erg}}^{\\text{MRC}} / B_w$) [bits/s/Hz]', fontsize=12)
plt.title('Capacidade Ergódica, Maximal Ratio Combining (MRC) sob Rayleigh[cite: 6]', fontsize=13)
plt.grid(True, which="both", ls="--", alpha=0.5)
plt.legend(fontsize=9, loc='upper left', ncol=2)
plt.xlim(-6, 21)

plt.tight_layout()
plt.show()
