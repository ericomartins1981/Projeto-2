import numpy as np
import matplotlib.pyplot as plt

N = 300_000
L_valores = [1, 2, 3, 4]

snr_db = np.linspace(-5, 20, 60)
snr_lin = 10**(snr_db / 10)

plt.figure(figsize=(9, 6))
cores = ['#1f77b4', '#ff7f0e', '#2ca02c', '#d62728']

rng = np.random.default_rng(seed=42)

for idx, L in enumerate(L_valores):
    capacidade_ergodica = []
    
    potencia_canal = rng.exponential(scale=1.0, size=(N, L))
    
    for snr_m in snr_lin:
        gamma_l = snr_m * potencia_canal
        
        gamma_s_egc = (1.0 / L) * (np.sum(np.sqrt(gamma_l), axis=1))**2
        
        cap = np.mean(np.log2(1 + gamma_s_egc))
        capacidade_ergodica.append(cap)
    
    plt.plot(
        snr_db,
        capacidade_ergodica,
        label=f'L = {L}',
        color=cores[idx],
        marker='+',
        linestyle='-',
        markevery=5,
        markersize=7,
        linewidth=0
    )

plt.title(f'Capacidade Ergódica - Equal Gain Combining (EGC) (Simulação)', fontsize=12)
plt.xlabel('SNR Média por Ramo $\\bar{\\gamma}$ (dB)', fontsize=11)
plt.ylabel('Capacidade Ergódica (bits/s/Hz)', fontsize=11)
plt.xlim([snr_db[0], snr_db[-1]])
plt.grid(True, which='both', linestyle='--', alpha=0.6)
plt.legend(title='Ordem de Diversidade', fontsize=10)
plt.tight_layout()
plt.show()
